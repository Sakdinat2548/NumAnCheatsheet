\section{Groups Recap + Euler $\varphi$}
\begin{sloppypar}\raggedright
\textbf{Group:} closure, identity $e$ (unique, $a{\circ}e{=}a$), associativity, inverses (unique). $(\mathbb Z,{+})$, $(\mathbb Z_n,{+})$ groups; $\mathbb Z_n^*$ group $\iff n$ prime. $|G|{=}\#$elem; $\mathrm{ord}(g)$ = smallest $n{>}0$ with $g^n{=}e$. \hl{Lagrange:} $\mathrm{ord}(g)\mid|G|$, $g^{|G|}{=}e$. \ex{Worked:} $|\mathbb Z_8^+|{=}8$, $\mathrm{ord}[1]{=}8$, $\mathrm{ord}[2]{=}4$, $\mathrm{ord}[7]{=}8$; gens $[1],[3],[5],[7]$.\\
$\varphi(n){=}\#\{1{\le}k{\le}n:\gcd{=}1\}$. \hl{Formulas:} $p$ prime $\varphi(p){=}p{-}1$; $\varphi(p^k){=}p^k{-}p^{k-1}$; coprime $\varphi(mn){=}\varphi(m)\varphi(n)$.\\
\ex{Worked:} $\varphi(16){=}8$ ($1,3,5,7,9,11,13,15$); $\varphi(17){=}16$.\\
$\begin{array}{l}
\varphi(14){=}\varphi(2)\varphi(7){=}(2{-}1)(7{-}1){=}1{\cdot}6{=}6\\
\varphi(9){=}9{-}3{=}6\ \text{(prime power)}
\end{array}$\\
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{Euler:} \hl{$a^{\varphi(n)}\equiv1\pmod n$ if $\gcd(a,n){=}1$}. FLT = prime case ($\varphi(p){=}p{-}1$) = Lagrange on $\mathbb Z_p^*$. \hl{Use:} prime $\to$ FLT (smaller exponent); composite $\to$ Euler. \hl{Idea:} reduce exponent mod $\varphi(n)$.
\end{kbox}
\begin{sloppypar}\raggedright
\ex{Worked $3^{347}\bmod14$:} Given $\gcd(3,14){=}1$, $\varphi(14){=}6$.\\
\wk{Step 1} (reduce exponent): $347{=}6{\cdot}57{+}5$ ($6{\cdot}57{=}342$).\\
\wk{Step 2} (evaluate): $3^5{=}243$; $243{-}17{\cdot}14{=}243{-}238{=}5\Rightarrow\hl{[5]}$. \wk{Check:} $\mathrm{pow}{=}5$.
\end{sloppypar}

\section{Complex Numbers}
\begin{sloppypar}\raggedright
Rect $z{=}a{+}bi$ (Re${=}a$, Im${=}b$), plane vector $(a,b)$. Polar $r(\cos\theta{+}i\sin\theta)$, $r{=}|z|{=}\sqrt{a^2{+}b^2}$. $|z|^2{=}z\bar z$, $\bar z{=}a{-}bi$.\\
$+,-$ = vector $+$ (parallelogram). Mult: $(a{+}bi)(c{+}di){=}(ac{-}bd){+}(ad{+}bc)i$.\\
\ex{Worked $3{+}4i$:} $|z|{=}5$, $z^{-1}{=}(3{-}4i)/25{=}0.12{-}0.16i$.\\
\ex{Worked $(1{+}6i)/(1{-}6i)$:} $(1{+}6i)^2/37{=}({-}35{+}12i)/37$.\\
${\approx}{-}0.9459{+}0.3243i$ \wk{Check} ($37{\cdot}({-}0.9459){=}{-}35$, $37{\cdot}0.3243{=}12$).\\
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{Euler:} \hl{$e^{i\theta}{=}\cos\theta{+}i\sin\theta$}; \textbf{DeMoivre:} \hl{$(re^{i\theta})^n{=}r^n(\cos n\theta{+}i\sin n\theta)$}.\\
\textbf{$n$-th roots:} \hl{$\sqrt[n]{r}(\cos(\theta{+}2k\pi)/n{+}i\sin(\theta{+}2k\pi)/n)$}, $k{=}0..n{-}1$.\\
\textbf{Div:} \hl{$z_1/z_2{=}z_1\bar z_2/|z_2|^2$}; \textbf{Inverse:} \hl{$z^{-1}{=}\bar z/|z|^2$}.\\
\textbf{Polar mult:} \hl{$r_1r_2e^{i(\theta_1+\theta_2)}$}: $e^{i\pi/3}{\cdot}e^{i\pi/6}{=}e^{i\pi/2}{=}i$.
\end{kbox}
\begin{sloppypar}\raggedright
\ex{Worked $8e^{i\pi/3}$:} $8(\frac12{+}i\frac{\sqrt3}{2}){=}4{+}4\sqrt3\,i{\approx}4{+}6.9282i$. \ex{Worked $11e^{3\pi i/2}$:} $11(0{-}i){=}{-}11i$.\\
Cube roots of $e^{i\pi/2}$: angles $\pi/6,5\pi/6,3\pi/2\Rightarrow\sqrt3/2{+}i/2,\ {-}\sqrt3/2{+}i/2,\ {-}i$.\\
\ex{Square roots of $e^{i\pi/4}$:} angles $\pi/8,9\pi/8$: $\cos\frac\pi8{=}\sqrt{2{+}\sqrt2}/2{\approx}0.9239$, $\sin\frac\pi8{=}\sqrt{2{-}\sqrt2}/2{\approx}0.3827$.\\
\ex{Rect$\to$polar:} $2{+}2i$: $r{=}\sqrt8{=}2\sqrt2$, $\theta{=}\pi/4\Rightarrow2\sqrt2\,e^{i\pi/4}$ (\wk{Check} $2\sqrt2(\sqrt2/2{+}i\sqrt2/2){=}2{+}2i$).\\
Geometry: ${\times}e^{i\theta}$ rotates by $\theta$ (lengths mult); $\bar z$ mirrors across real axis; unit circle $\{|z|{=}1\}$ subgroup (closed $|zw|{=}1$, inv $\bar z$). \wk{Plot recipe:} mark Re on $x$, Im on $y\to$ arrow; add tip-to-tail; mult adds angles, mults lengths.\\
Groups: $(\mathbb C,{+})$ $e{=}0$ inv ${-}z$; $(\mathbb C^*,{\cdot})$ $e{=}1$ inv $\bar z/|z|^2$; $C_4^*{=}\{\pm1,\pm i\}$ gen $i$; $C_8^*$ gen $(1{+}i)/\sqrt2$; $\{1,{-}1\}$ group (gen ${-}1$); $\{i,{-}i\}$ \hl{not} ($i{\cdot}i{=}{-}1$ missing). \end{sloppypar}

\section{Roots of Unity}
\begin{sloppypar}\raggedright
Powers $\omega^0..\omega^{n-1}$ = all $n$ roots cyclically (regular $n$-gon on unit circle, angle $2\pi/n$ apart); primitive = generator; $n$ distinct roots of $z^n{-}1{=}0$.\\
\hl{Primitive:} $\omega_n^k$ generates all $\iff\gcd(k,n){=}1$; e.g. $n{=}8$: $k{=}1,3,5,7$. \wk{Why:} $\mathrm{ord}(\omega_n^k){=}n/\gcd(k,n)$ — shared factors shortcut the cycle (e.g. $\omega_8^2{=}\omega_4$, order $4$). \wk{Gen test:} $g^d{\ne}1$ for all proper divisors $d{\mid}n$ (Z13: $2^1,2^2,2^3,2^4,2^6{=}2,4,8,3,12{\ne}1$).\\
\hl{Cyclic:} $C_n^*{=}\langle\omega_n\rangle$; reduce first: $\omega_8^2{=}\omega_4{=}i$.
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{Root:} \hl{$\omega_n{=}e^{2\pi i/n}$}.\\
\textbf{Relations ($n$ even; last two $\forall n$):} $\omega^n{=}1$, $\omega^{n/2}{=}{-}1$, $\omega^2{=}\omega_{n/2}$, $\omega^{-1}{=}\bar\omega$, $\omega^{k+n/2}{=}{-}\omega^k$, $\omega^{2k}{=}\omega_{n/2}^k$, $\omega^{n-k}{=}\bar\omega^k$.
\end{kbox}
\begin{sloppypar}\raggedright
\ex{Worked $\omega_8{=}(1{+}i)/\sqrt2$:} $\omega_8^4{=}{-}1$, $\omega_8^8{=}1$, $\omega_8^{-5}{=}\omega_8^{-5+8}{=}\omega_8^3{=}({-}1{+}i)/\sqrt2$; $\omega_{16}^4{=}e^{8\pi i/16}{=}i{=}\omega_4$. $\sqrt{i}{=}(1{+}i)/\sqrt2$ (\wk{Check} $((1{+}i)/\sqrt2)^2{=}i$); $\sqrt{-i}{=}({-}1{+}i)/\sqrt2$.\\
\wk{Eval trick:} reduce exponents mod $n$, use $\omega_2{=}{-}1$.\\
\ex{Worked} $p{=}x^4{-}3x^3{+}x^2{-}2x{-}1$ at $\omega_4{=}i$: powers $\omega_4^2{=}{-}1$, $\omega_4^3{=}{-}i$, $\omega_4^4{=}1$.\\
Then $1{-}3({-}i){+}({-}1){-}2i{-}1{=}1{+}3i{-}1{-}2i{-}1{=}{-}1{+}i$.\\
Lecture: $3\omega_4^3{-}2\omega_4^2{+}2{+}\omega_4^8{=}3({-}i){+}2{+}2{+}1{=}5{-}3i$.
\end{sloppypar}

\section{Iso $C_n^*\cong\mathbb Z_n^+$}
\begin{sloppypar}\raggedright
\hl{Map $\omega_n^k\leftrightarrow[k]_n$;} ${\times}\leftrightarrow{+}$: $\omega^k\omega^l{=}\omega^{k+l}\leftrightarrow[k]{+}[l]{=}[k{+}l]$. Gen $\leftrightarrow$ gen. \hl{Iso:} bijection preserving op ($C_4^*$ table ${\equiv}$ $\mathbb Z_4^+$ table). \hl{Subgroup:} subset + closed under op.\\
\ex{Worked $C_4\leftrightarrow\mathbb Z_4$:} $[0]{\leftrightarrow}1,[1]{\leftrightarrow}i,[2]{\leftrightarrow}{-}1,[3]{\leftrightarrow}{-}i$; $[1]{+}[2]{=}[3]\leftrightarrow i{\cdot}({-}1){=}{-}i$; $[3]{+}[2]{=}[5]{=}[1]\leftrightarrow({-}i)({-}1){=}i$. Tables same shape $\Rightarrow$ iso.\\
\wk{No-iso test:} $|G|$ must match + generator orders; e.g. $|C_2^*|{=}2$ vs $|\mathbb Z_4^+|{=}4\Rightarrow$ no iso. \ex{Fill $C_5^*$ via iso:} $\omega_5^3{\cdot}\omega_5^4{=}\omega_5^7{=}\omega_5^2\leftrightarrow[3]{+}[4]{=}[7]{=}[2]$. Subgroups: $C_{2^m}^*{\subset}C_{2^{m+1}}^*$ ($z^{2^m}{=}1\Rightarrow(z^{2^m})^2{=}1$).
\end{sloppypar}

\section{Poly: FTA, Fields, Coeff/Value}
\begin{sloppypar}\raggedright
\textbf{Field:} $+,-,{\times},/$ (no $/0$); $\mathbb Q,\mathbb R,\mathbb C$ yes; $\mathbb Z$ no ($a/b$ can leave $\mathbb Z$); $\mathbb N$ no ($a{-}b$ can leave $\mathbb N$); $\mathbb Z_p$ ($p$ prime) yes (every ${\ne}0$ invertible). Polys over field $K$ (usually $\mathbb R/\mathbb C$); some fields give no roots. Alg closed: every $\deg{\ge}1$ has a root; $\mathbb C$ is (FTA); $\mathbb R$ is \hl{not} ($x^2{+}1{=}0$ no real root).\\
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{FTA:} deg $n$ $\Rightarrow$ \hl{exactly $n$ roots in $\mathbb C$} counting mult: $f{=}a_n\prod(x{-}x_i)$.
\end{kbox}
\begin{sloppypar}\raggedright
\ex{Worked $x^4{+}16{=}0$:} no real roots; 4 in $\mathbb C$; two: $\sqrt2{+}i\sqrt2$, ${-}\sqrt2{+}i\sqrt2$ (\wk{Check} $(\sqrt2{+}i\sqrt2)^2{=}4i$, $^2{=}{-}16$). All four: $\pm\sqrt2{\pm}i\sqrt2$ (signs independent; $|x|{=}2$, angles $\pi/4,3\pi/4,5\pi/4,7\pi/4$).\\
\ex{4th roots of $-1$:} ${-}1{=}e^{i\pi}$: angles $(\pi{+}2k\pi)/4{=}\pi/4,3\pi/4,5\pi/4,7\pi/4\Rightarrow(\pm1{\pm}i)/\sqrt2$ = odd $\omega_8$ powers (all primitive).\\
\wk{Proof (no real roots):} for real $x$, $x^4{\ge}0$ (even power), so $x^4{+}16{\ge}16{>}0$ — never $0$. I.e. $x^4{=}{-}16$ impossible over $\mathbb R$ (LHS ${\ge}0$). Hence all 4 FTA roots are nonreal. Factor thm: $f(a){=}0\iff(x{-}a)\mid f$; $(x{-}1)^2{=}x^2{-}2x{+}1$ (mult 2). Note: $5$ prime in $\mathbb Z$ but $5{=}(2{-}i)(2{+}i)$ in $\mathbb Z[i]$.\\
\textbf{Coeff} $(a_0..a_n)$; \hl{value:} $\deg{\le}n$ unique from $n{+}1$ pts (Vandermonde $\det{\ne}0$). \hl{Product deg $n{+}m$ needs $n{+}m{+}1$ pts;} pad zeros ($4{+}5x\to4{+}5x{+}0x^2$). Pointwise $h(x_i){=}f(x_i)g(x_i)$, $O(m)$; schoolbook $O(n^2)$.\\
\ex{Worked $f{=}1{-}x,g{=}2{+}x$ at ${-}2,0,1$:} $f{:}1{-}({-}2){=}3,1,0$; $g{:}2{+}({-}2){=}0,2,3$; $h{:}0,2,0\Rightarrow h{=}2{-}x{-}x^2$ (\wk{Check} $2{+}2{-}4{=}0$, $2$, $2{-}1{-}1{=}0$).\\
\wk{Sketch recipe:} mark roots $\to$ lead-coeff sign for ends $\to$ $y$-intercept $\to$ join; graphical mult: sample ${\ge}6$ $x$'s, multiply $y$'s, plot.\\
\wk{Point-count recipe:} single deg-$n$ $\to$ $n{+}1$ pts; product deg $n{+}m$ $\to$ $n{+}m{+}1$; FFT $\to$ next pow2 ${\ge}$ that.
\end{sloppypar}

\section{FFT/IFFT + Costs}
\begin{sloppypar}\raggedright
Even/odd: $p(x){=}p_e(x^2){+}x\,p_o(x^2)$. \ex{Worked} $1{-}2x{+}3x^2{-}4x^3{+}5x^4{-}6x^5{+}7x^6{-}8x^7$: $p_e(y){=}1{+}3y{+}5y^2{+}7y^3$, $p_o(y){=}{-}2{-}4y{-}6y^2{-}8y^3$; depth of recursion tree ${=}3$ ($8{=}2^3$).
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{Butterfly} ($k{=}0..n/2{-}1$): \hl{$p(\omega^k){=}p_e(\omega^{2k}){+}\omega^k p_o(\omega^{2k})$; $p(\omega^{k+n/2}){=}p_e(\omega^{2k}){-}\omega^k p_o(\omega^{2k})$}. Needs $n{=}2^t$.
\end{kbox}
\begin{sloppypar}\raggedright
\ex{Butterfly $n{=}4$ ($\omega{=}i$):}\\
$\begin{array}{ll}
k{=}0: & p(1){=}p_e(1){+}p_o(1)\\
& p({-}1){=}p_e(1){-}p_o(1)\\
k{=}1: & p(i){=}p_e({-}1){+}i\,p_o({-}1)\\
& p({-}i){=}p_e({-}1){-}i\,p_o({-}1)
\end{array}$\\
\hl{$N_{\mathrm{FFT}}\ge2n{-}1$}, smallest pow2 (4+4 coeffs deg3: product deg6, 7 coeffs $\Rightarrow \hl{N{=}8}$). Pipeline: coeff$\to$FFT$\to$vals$\to$pointwise ${\times}\to$IFFT$\to$coeff. Zero-pad; extra points OK if same poly. \wk{Idea:} one $(p_e,p_o)$ pair gives two values (halves work).\\
\end{sloppypar}
\begin{kbox}\raggedright
\textbf{IFFT:} $a_j{=}\frac{1}{n}\sum_{k=0}^{n-1}y_k(\omega^{-j})^k$, $j{=}0..n{-}1$.
\end{kbox}
\begin{sloppypar}\raggedright
\hl{Costs:} FFT/IFFT $O(n\log n)$ (depth $\log n\times O(n)$/level); pointwise $O(n)$; total $O(n\log n)$.\\
\ex{Worked FFT $q{=}1{-}x^2{+}4x^3$} ($N{=}4$, $\omega_4{=}i$): $q(1){=}4$; $q(i){=}2{-}4i$; $q({-}1){=}{-}4$; $q({-}i){=}2{+}4i$. Butterfly check: $p_e(y){=}1{-}y$, $p_o(y){=}4y$; $q(i){=}p_e({-}1){+}i\,p_o({-}1){=}2{+}i({-}4){=}2{-}4i$.\\
\ex{Butterfly table:} $k{=}0$: $p_e(1){=}0$, $p_o(1){=}4\Rightarrow q(1){=}0{+}4{=}4$, $q({-}1){=}0{-}4{=}{-}4$.\\
$k{=}1$: $p_e({-}1){=}2$, $p_o({-}1){=}{-}4\Rightarrow q(i){=}2{+}i({-}4){=}2{-}4i$, $q({-}i){=}2{-}i({-}4){=}2{+}4i$.\\
\ex{Full pipeline} $f{=}1{-}x,g{=}2{+}x$ ($N{=}4$): $F{:}0,1{-}i,2,1{+}i$; $G{:}3,2{+}i,1,2{-}i$; $H{=}F{\cdot}G{:}0,3{-}i,2,3{+}i$.\\
\ex{IFFT:} $a_0{=}(0{+}3{-}i{+}2{+}3{+}i)/4{=}8/4{=}2$; $a_1{=}({-}3i{-}1{-}2{+}3i{-}1)/4{=}{-}1$; $a_2{=}{-}1\Rightarrow$\hl{$h{=}2{-}x{-}x^2$} \wk{Check} OK.\\
\ex{Worked IFFT $y{=}({-}1,i,1,2)$:} $(\omega^{-j})^k$ ($k{=}0..3$, $\omega{=}i$):\\
$j{=}0$: $1,1,1,1$; $j{=}1$: $1,{-}i,{-}1,i$; $j{=}2$: $1,{-}1,1,{-}1$; $j{=}3$: $1,i,{-}1,{-}i$.\\
$a_0{=}(y_0{+}y_1{+}y_2{+}y_3)/4$\\
${=}({-}1{+}i{+}1{+}2)/4{=}(2{+}i)/4{=}0.5{+}0.25i$.\\
$a_1{=}(y_0{+}y_1({-}i){+}y_2({-}1){+}y_3(i))/4$\\
${=}({-}1{+}1{-}1{+}2i)/4{=}({-}1{+}2i)/4{=}{-}0.25{+}0.5i$ ($i{\cdot}({-}i){=}1$).\\
$a_2{=}(y_0{+}y_1({-}1){+}y_2{+}y_3({-}1))/4$\\
${=}({-}1{-}i{+}1{-}2)/4{=}({-}2{-}i)/4{=}{-}0.5{-}0.25i$.\\
$a_3{=}(y_0{+}y_1(i){+}y_2({-}1){+}y_3({-}i))/4$\\
${=}({-}1{-}1{-}1{-}2i)/4{=}({-}3{-}2i)/4{=}{-}0.75{-}0.5i$ ($i{\cdot}i{=}{-}1$).\\
\wk{Check} $p(i){=}a_0{+}a_1i{-}a_2{-}a_3i{=}i{=}y_1$.\\
\ex{Write $p$:} $p(x){=}(0.5{+}0.25i){+}({-}0.25{+}0.5i)x$\\
${+}({-}0.5{-}0.25i)x^2{+}({-}0.75{-}0.5i)x^3$.
\end{sloppypar}

\section{Int Mult, GCD, Systems}
\begin{sloppypar}\raggedright
Int as poly base $B$: $325\to5{+}2x{+}3x^2$ ($p(10){=}325$). Steps: polys $\to$ FFT $\to$ pointwise ${\times}$ $\to$ IFFT $\to$ carry to $[0,B{-}1]$ $\to$ eval $B$. \hl{Carry:} $[2,15,7]\to[2,5,8]{=}852$ (raw $2{+}150{+}700{=}852$; notes typo $825$). $5678\to8{+}7x{+}6x^2{+}5x^3$; $q(10){=}5678$, $q(2){=}86$. $d$ decimals $\to\lceil d/3\rceil$ base-$1000$ digits: $120!$: $199\to67$ ($66{\cdot}3{=}198{<}199$). \hl{Idea:} every digit must land in $[0,B{-}1]$ (carry the rest).\\
