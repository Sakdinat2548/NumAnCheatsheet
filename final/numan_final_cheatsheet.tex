\section{Regression / Data Fitting (Th.1--6)}
\begin{sloppypar}\raggedright
\textbf{Regression $\ne$ interpolation:} interpolation trusts every measurement and passes through ALL points; regression assumes noise and finds the best OVERALL trend, passing through (almost) none.
Deviations $d_i=y_i-\phi(x_i)$, data $(x_i,y_i)$, $i=0..n$ ($n{+}1$ points).
\textbf{Error choices:} naive $S=\sum d_i$ (rejected: $+,-$ cancel); abs $S=\sum|d_i|$ (rejected: kink at 0, no derivative); \hl{squared} $S=\sum d_i^2$ (chosen: smooth everywhere, so ``set gradient $=0$'' gives solvable linear equations; big misses punished quadratically).
Example (Lec14): pts $(1,2),(2,1)$, line through $(1,1),(2,2)$: $d_0=1,d_1=-1$: naive $S=0$ (cancels!), abs $S=2$, squared $S=2$.

\subsection{Linear regression}
$\hat y=a_0+a_1x$, $S=\sum(y_i-a_0-a_1x_i)^2$. General fit $\phi(x,a_0{..}a_\ell)$: $\tfrac{\partial S}{\partial a_k}=-2\sum_{i=0}^n(y_i-\phi(\cdots))\tfrac{\partial\phi}{\partial a_k}=0$, $k=0..\ell$ ($\ell{+}1$ equations). For the line this gives the two normal equations below.
$\partial_{a_0}$: $\sum y_i-(n{+}1)a_0-a_1\sum x_i=0$; $\partial_{a_1}$: $\sum x_iy_i-a_0\sum x_i-a_1\sum x_i^2=0$.\\
Divide by $n{+}1$ $\Rightarrow$ $a_0+a_1\bar x=\bar y$, $a_0\bar x+a_1\overline{x^2}=\overline{xy}$; Cramer gives
\begin{kbox}
\centering
$\displaystyle a_1=\frac{\overline{xy}-\bar x\bar y}{\overline{x^2}-\bar x\cdot\bar x},\qquad a_0=\bar y-a_1\bar x$
\end{kbox}
with $\bar x=\tfrac1{n+1}\sum x_i$, $\overline{x^2}=\tfrac1{n+1}\sum x_i^2$, $\overline{xy}=\tfrac1{n+1}\sum x_iy_i$.\\
\hl{Trap:} $\overline{x^2}\ne\bar x^2$; denominator $=0$ iff all $x_i$ equal (no fit possible).

\subsection{Hessian + minimum test}
\begin{kbox}
\centering
$\displaystyle H=\begin{pmatrix}S_{a_0a_0}&S_{a_0a_1}\\S_{a_1a_0}&S_{a_1a_1}\end{pmatrix}$\\[1pt]
$\displaystyle S_{a_0a_0}=2(n{+}1),\ S_{a_0a_1}=2\sum x_i,\ S_{a_1a_1}=2\sum x_i^2$\\[1pt]
$\displaystyle \det H=4\big[(n{+}1)\sum x_i^2-(\sum x_i)^2\big]$; min $\iff S_{a_0a_0}>0$ AND $\det H>0$
\end{kbox}
Intuition: $H$ = curvature of the error bowl; positive-definite = bowl opens upward = exactly one bottom. \textbf{Cauchy--Schwarz} $\Rightarrow\det H\ge0$; $>0$ if $\ge2$ distinct $x_i$ $\Rightarrow$ unique global min (strictly convex).

\subsection{Multivariable + pseudoinverse}
$\hat y=a_0+a_1x_1+\cdots+a_mx_m$, $X$ is $(n{+}1)\times(m{+}1)$ with first col of 1's. $S=\|\bar y-X\bar a\|^2$.
\begin{kbox}
\centering
$\displaystyle X^TX\bar a=X^T\bar y,\qquad \bar a=\underbrace{(X^TX)^{-1}X^T}_{X^+}\bar y$
\end{kbox}
$X^+=$ Moore--Penrose pseudoinverse ($=X^{-1}$ when $X$ square). Idea: $X^TX\bar a=X^T\bar y$ projects $\bar y$ onto the column space of $X$ (closest achievable fit); $X^+$ is that projection written as a formula. \hl{Trap:} $X^+X=I$ holds; $XX^+=I$ generally does NOT.

\subsection{ML: classification}
Features = measured inputs $(x_1,x_2)$; model outputs a score; plane $a_0+a_1x_1+a_2x_2=0\cap x_1x_2$-plane = \hl{decision boundary} (class 1 vs 2). Logistic $\sigma(z)=\tfrac1{1+e^{-z}}: \mathbb R\to[0,1]$ squashes the score to a probability; classifier $\sigma(a_0+a_1x_1+a_2x_2)$.
\end{sloppypar}

\section{Linearization}
\wk{Goal:} transform $(\tilde x,\tilde y)$ so model is linear; fit; inverse-transform back. Why logs work: $\ln$ turns $\times$ into $+$ and powers into $\times$ (e.g. $\ln(Ae^{Bx})=\ln A+Bx$). Steps: (1) change vars; (2) fit $\tilde y=a_0+a_1\tilde x$; (3) recover $A,B$; (4) check fit.
\begin{itemize}
\item $y=Ae^{Bx}$: $\tilde x=x$, $\tilde y=\ln y$; $a_0=\ln A$, $a_1=B$ $\Leftrightarrow$ $A=e^{a_0}$.
\item $y=Ax^n$: $\tilde x=\ln x$, $\tilde y=\ln y$; $a_0=\ln A$, $a_1=n$.
\item $y=A+\tfrac{B}{x}$: $\tilde x=\tfrac1x$, $\tilde y=y$; $a_0=A$, $a_1=B$.
\item $y=\tfrac1{A+Bx}$: $\tilde x=x$, $\tilde y=\tfrac1y$; $a_0=A$, $a_1=B$.
\item $y=\tfrac{Ax}{B+x}$: $\tilde x=\tfrac1x$, $\tilde y=\tfrac1y$; $a_0=\tfrac1A$, $a_1=\tfrac{B}{A}$ $\Leftrightarrow$ $A=\tfrac1{a_0}$, $B=\tfrac{a_1}{a_0}$.
\end{itemize}
\hl{Trap:} not everything linearizes (e.g. $a_0e^{-x}\sin(a_1+2\pi x)$). \wk{Model from graph:} straight $\to$ linear; explosive growth $\to Ae^{Bx}$; power curve $\to Ax^n$; saturating hyperbola $\to Ax/(B+x)$; reciprocal decay $\to A+B/x$.

\section{Grids + Riemann Sums}
\begin{sloppypar}\raggedright
Partition $a=x_0<\cdots<x_n=b$, $h_i=x_i-x_{i-1}$, sample $x_i^*\in[x_{i-1},x_i]$: $\int_a^bf\approx\sum f(x_i^*)h_i$.
\textbf{Uniform:} $h_i\equiv h=\tfrac{b-a}{n}$, $n{+}1$ points. \textbf{Non-uniform:} varying $h_i$ (adaptive: fine where $f$ changes fast, coarse elsewhere). Idea: spend grid points where the function wiggles; a uniform grid wastes them where $f$ is flat and starves them where $f$ is steep.
\textbf{Riemann integral} $=\lim_{\max h_j\to0}\sum h_if(x_i^*)$; $f$ bounded, at most countably many jumps (isolated jumps are OK, the area under one point is zero).
\end{sloppypar}

\section{Rectangle / Trapezoid / Simpson}
\begin{sloppypar}\raggedright
\textbf{Sample point:} $x_i^*=\alpha x_{i-1}+(1-\alpha)x_i$: $\alpha=1$ left, $\alpha=0$ right, $\alpha=\tfrac12$ midpoint. (Quiz7: $0.2x_{i-1}+0.8x_i$ is $\alpha=0.2$; Lec16: $x_{i-1}+h_i/3$ is $\alpha=\tfrac23$, $x_{i-1}+2h_i/3$ is $\alpha=\tfrac13$.)
General non-uniform: left $\sum h_if(x_{i-1})$, right $\sum h_if(x_i)$, mid $\sum h_if(\tfrac{x_{i-1}+x_i}2)$.
\begin{kbox}
\centering
$\displaystyle \tilde I_{\text{Tr}}=\tfrac12\sum_{i=1}^n\big(f(x_{i-1})+f(x_i)\big)h_i$
\end{kbox}
Uniform trapezoid: $\tfrac{h(f_0+f_n)}2+h\sum_{i=1}^{n-1}f_i$.
\wk{Trap idea:} one strip = rectangle $hQ_1$ + triangle $\tfrac h2(Q_2-Q_1)$ $=h(Q_1+Q_2)/2$; uniform version averages neighbors (interior $f_i$ counted twice $\to$ single sum).
\textbf{Simpson} ($n=2m$ even, uniform): fits parabola on each pair. Why $O(h^4)$? A parabola matches $f$'s value, slope AND curvature, so the leftover is two orders smaller than trapezoid (which matches only value + slope). Cost: more $f$-evaluations per step.
\begin{kbox}
\centering
$\displaystyle I_{\text{parab}}=\tfrac{h}{3}\big\{f_0+f_n+4\sum_{\text{odd }i}f_i+2\sum_{\text{even }i<n}f_i\big\}$
\end{kbox}
\wk{Sketch:} rectangles = bars at $x_i^*$; trapezoids = straight segments node-to-node; Simpson = parabola arcs per pair (blank axes: mark $n{+}1$ nodes first). Simpson weights: endpoints $\times1$, odd indices $\times4$, even interior $\times2$.

\subsection{Accuracy estimates}
\begin{itemize}
\item left/right rectangle $O(h)$; midpoint $O(h^2)$; trapezoid $O(h^2)$; Simpson $O(h^4)$.
\item Read $O(h^p)$ as: halving $h$ divides the error by $2^p$ (rect $\div2$, mid/trap $\div4$, Simpson $\div16$).
\item If $|h|<1$: midpoint beats left/right; trap $\approx$ midpoint order.
\item Left-rect global: $|E|\le\tfrac{M_1(b-a)}2h$, $M_1=\max|f'|$. Midpoint: $|E|\le\tfrac{M_2(b-a)}{24}h^2$. (Taylor + symmetry kills $f'$ term.)
\item No elementary antiderivative ($\int e^{x^2}dx$, $\int\tfrac{dx}{\ln x}$): cross-check Simpson vs trapezoid.
\end{itemize}
\end{sloppypar}

\section{ODEs: IVP + Numeric Methods}
\begin{sloppypar}\raggedright
$y'=f(x,y)$: slope field tells the direction at every point; a solution follows the arrows. General solution $y(x,C)$ = whole family of curves; IVP (Cauchy) $+y(x_0)=y_0$ picks the ONE curve through $(x_0,y_0)$ (fix $C$ by plugging IC; existence/uniqueness assumed).
\hl{Check-solution trap:} differentiate candidate AND plug into ODE — both must hold for ALL $C$.
\wk{Bridge (Lec19):} Euler comes from the finite difference $y'\approx\tfrac{y_{i+1}-y_i}{h}$: replace $y'$ in $y'=f(x,y)$ and solve for $y_{i+1}$.
\begin{kbox}
\centering
$\displaystyle y_{i+1}=y_i+hf(x_i,y_i),\quad x_{i+1}=x_i+h\quad\text{(Euler)}$
\end{kbox}
\begin{kbox}
\centering
$\displaystyle y_{i+1}=y_i+hf(x_{i+\frac12},y_{i+\frac12}),\quad x_{i+\frac12}=x_i+\tfrac h2,\ y_{i+\frac12}=y_i+\tfrac h2f(x_i,y_i)$
\end{kbox}
\textbf{RK4:} weighted average of 4 trial slopes with weights $(1,2,2,1)/6$: $k_1=f(x_n,y_n)$; $k_2=f(x_n{+}\tfrac h2,y_n{+}\tfrac h2k_1)$; $k_3=f(x_n{+}\tfrac h2,y_n{+}\tfrac h2k_2)$; $k_4=f(x_n{+}h,y_n{+}hk_3)$.\\
$y_{n+1}=y_n+\tfrac h6(k_1{+}2k_2{+}2k_3{+}k_4)$. Price: 4 $f$-evaluations per step for $O(h^4)$.
\textbf{Global errors:} Euler $O(h)$ (local $O(h^2)\times N\sim1/h$); midpoint $O(h^2)$; RK4 $O(h^4)$ (4 evals/step).
\wk{Euler error idea:} Taylor $y(x{+}h)=y(x)+hy'(x)+\tfrac{h^2}2y''(\xi)$; one Euler step from exact $y(x_i)$ misses by $O(h^2)$ (local); $N=(b-x_0)/h=O(1/h)$ steps accumulate to $O(h)$ global.
\wk{Midpoint idea:} predict midpoint slope with one Euler half-step, then step full $h$ along it; symmetric $\Rightarrow$ local $O(h^3)$, global $O(h^2)$.
\end{sloppypar}

\section{Linear Algebra* (starred)}
\begin{sloppypar}\raggedright
$\det=$ volume scale factor of the matrix map; $\det\ne0\iff$ invertible ($\det=0$ = space collapses). $\det(AB)=\det A\det B$. $Ax=b$ has a unique solution $\iff\det A\ne0\iff n$ pivots (else $0$ or $\infty$ solutions).\\
\ex{Basic ops:} $\big(\begin{smallmatrix}1&2\\3&4\end{smallmatrix}\big)\big(\begin{smallmatrix}5\\6\end{smallmatrix}\big)=\big(\begin{smallmatrix}1\cdot5+2\cdot6\\3\cdot5+4\cdot6\end{smallmatrix}\big)=\big(\begin{smallmatrix}17\\39\end{smallmatrix}\big)$ (row $\cdot$ column).\\
\textbf{Laplace:} expand along row/col with $(-1)^{i+j}$ cofactors (pick the row/col with most zeros). Elimination: row ops change det by sign/scale (swap $\to$ $-$, scale row by $c$ $\to$ $\times c$); $\det=$ product of $U$ diagonal.\\
\textbf{LU:} $A=LU$ records elimination ($L$ = unit-lower multipliers, $U$ = upper result); solve $Ly=b$, then $Ux=y$ — two cheap triangular passes instead of re-eliminating. \ex{Worked:} $A=\big(\begin{smallmatrix}2&1&0\\1&3&1\\0&2&4\end{smallmatrix}\big)$: $R_2-\tfrac12R_1$, $R_3-\tfrac45R_2'$ $\to$ $U$ diag $(2,2.5,3.2)$, $\det=2\cdot2.5\cdot3.2=16$.
\end{sloppypar}

\section{Midterm Recap (compact)}
\begin{sloppypar}\raggedright
\begin{itemize}
\item \textbf{Taylor:} $f(x)=\sum_{k=0}^n\tfrac{f^{(k)}(x_0)}{k!}(x-x_0)^k+R_n$, $R_n=\tfrac{f^{(n+1)}(\xi)}{(n+1)!}(x-x_0)^{n+1}$, $|R_n|\le\tfrac{Md^{n+1}}{(n+1)!}$.
\item Odd fn: only odd powers (deg 8 $\to x^7$); index $R_{2m+1}$, bound $\tfrac{12^{2m+2}}{(2m+2)!}$.
\item \textbf{Maclaurin:} $e^x,\sin,\cos$ converge $\forall\mathbb R$; $\tfrac1{1-x}=\sum x^k$, $\ln(1{+}x)$, $\arctan$ for $|x|<1$.
\item \textbf{Horner:} tableau at $c$ gives $p(c)$ (last) + quotient $q$; tableau on $q$ gives $p'(c)$.
\item \textbf{Interpolation:} $\ell_i(x)=\prod_{j\ne i}\tfrac{x-x_j}{x_i-x_j}$, $p=\sum y_i\ell_i$; err $=\tfrac{f^{(n+1)}(\xi)}{(n+1)!}\prod(x-x_i)$ ($=0$ at nodes).
\item Vandermonde $\det=\prod_{i<j}(x_j-x_i)\ne0$ (distinct nodes); DD: $f[x_i,x_{i+1}]=\tfrac{\Delta f}{\Delta x}$, Newton $p_n=f[x_0]+\sum f[x_0{..}x_k]\prod_{j<k}(x-x_j)$.
\end{itemize}
\end{sloppypar}

\subsection{Coffee regression (Quiz 6)}
Given $x:0,1,2,3$; $y:3,5,8,7$ ($n{+}1{=}4$ pts).
Step 1 — sums (add term by term): $\sum x_i=0{+}1{+}2{+}3=6$; $\sum y_i=3{+}5{+}8{+}7=23$; $\sum x_i^2=0{+}1{+}4{+}9=14$; $\sum x_iy_i=0{+}5{+}16{+}21=42$.
Step 2 — means (divide by 4): $\bar x=6/4=1.5$; $\bar y=23/4=5.75$; $\overline{x^2}=14/4=3.5$; $\overline{xy}=42/4=10.5$.
Step 3 — coeffs (plug into boxed formula): $a_1=\tfrac{10.5-1.5\cdot5.75}{3.5-1.5^2}$\\
$=\tfrac{10.5-8.625}{3.5-2.25}=\tfrac{1.875}{1.25}=1.5$; $a_0=5.75-1.5\cdot1.5$\\
$=5.75-2.25=3.5$. $\boxed{\hat y=3.5+1.5x}$.
Step 4 — predict $x{=}4$: $\hat y=3.5+1.5\cdot4=3.5+6=9.5$ (mark $(4,9.5)$ on the scatter plot with the line).
\wk{Plot recipe (Quiz6 Q2):} scatter the 4 data points, draw the line through $(0,3.5)$ and $(4,9.5)$ — regression line need NOT pass through data points.
Step 5 — check fit: $\hat y:3.5,5.0,6.5,8.0$ vs $y:3,5,8,7$; residuals $-0.5,0,1.5,-1.0$; squared error $S=0.25+0+2.25+1=3.5$ (this $S$ is what least squares minimized).
\ex{Battery prediction:} at $t{=}70$: $0.714\cdot70+3.57=53.55\%$ (extrapolation — flag as less reliable).
\ex{Hessian:} from Step 1 sums ($\sum x_i=6$, $\sum x_i^2=14$): $S_{a_0a_0}=2\cdot4=8$, $S_{a_0a_1}=2\cdot6=12$, $S_{a_1a_1}=2\cdot14=28$;\\
$\det H=8\cdot28-12^2=224-144=80>0$ (+ $8>0$) $\Rightarrow$ min.

\subsection{Battery fit (Lec 14)}
Given $t:0,10,20,30,40,50,60$; charge $5,10,15,25,35,40,45$ ($n{+}1{=}7$).
Sums: $\sum t_i=0{+}10{+}20{+}30{+}40{+}50{+}60=210$; $\sum y_i=5{+}10{+}15{+}25{+}35{+}40{+}45=175$; $\sum t_i^2=0{+}100{+}400{+}900{+}1600{+}2500{+}3600=9100$; $\sum t_iy_i=0{+}100{+}300{+}750{+}1400{+}2000{+}2700=7250$.
Means (divide by 7): $\bar x=210/7=30$; $\bar y=175/7=25$; $\overline{x^2}=9100/7=1300$; $\overline{xy}=7250/7\approx1035.71$.
Coeffs: $a_1=\tfrac{1035.71-30\cdot25}{1300-30^2}=\tfrac{1035.71-750}{1300-900}$\\
$=\tfrac{285.71}{400}\approx0.714$; $a_0=25-0.714285\cdot30$\\
$=25-21.43\approx3.57$. $\boxed{\phi(x)=0.714x+3.57}$.
Check fit: $\hat y:3.57,10.71,17.85,24.99,32.13,39.27,46.41$ vs $y:5,10,15,25,35,40,45$\\
(residuals $1.43,-0.71,-2.85,0.01,2.87,0.73,-1.41$).

\subsection{Exp linearization (HW09)}
Given $x:1,2,3,5$, $y:8,4,2,1$; model $y=Ae^{Bx}$ $\Rightarrow$ transform $\tilde x=x$, $\tilde y=\ln y$.
Step 0 — transform each $y$: $\ln8\approx2.07944$, $\ln4\approx1.38629$, $\ln2\approx0.69315$, $\ln1=0$. So $\tilde x:1,2,3,5$; $\tilde y:2.07944,1.38629,0.69315,0$.
Step 1 — means ($n{+}1{=}4$): $\bar{\tilde x}=(1{+}2{+}3{+}5)/4=11/4=2.75$; $\bar{\tilde y}=(2.07944{+}1.38629{+}0.69315{+}0)/4=4.15888/4\approx1.03972$ (note $=\ln64/4=1.5\ln2$); $\overline{\tilde x^2}=(1{+}4{+}9{+}25)/4=39/4=9.75$; $\overline{\tilde x\tilde y}=(1\cdot2.07944{+}2\cdot1.38629{+}3\cdot0.69315{+}5\cdot0)/4=6.93147/4\approx1.73287$ (note $=2.5\ln2$).
Step 2 — fit: $a_1=\tfrac{1.73287-2.75\cdot1.03972}{9.75-2.75^2}=\tfrac{1.73287-2.85923}{9.75-7.5625}$\\
$=\tfrac{-1.12636}{2.1875}\approx-0.5149$; $a_0=1.03972-(-0.5149)\cdot2.75$\\
$=1.03972+1.41598\approx2.4557$.
Step 3 — back-transform ($A=e^{a_0}$, $B=a_1$): $A=e^{2.4557}\approx11.65$, $B\approx-0.515<0$. Check: $\hat y=11.65e^{-0.515x}$ gives $7.0,4.2,2.5,0.9$ vs $8,4,2,1$ — reasonable.
\ex{Nonlinear example (Lec15):} $x:1,2,3,4$, $y:0.45,0.81,1.05,1.13$ for $y=\tfrac{Ax}{B+x}$: $\tilde x=1/x$, $\tilde y=1/y$ give $a_1\approx1.834$, $a_0\approx0.368$,\\
so $A=1/a_0\approx2.71$, $B=a_1/a_0\approx4.98$.

\subsection{Trapezoid $\sin x$ (Quiz 7)}
Given $[0,3\pi/2]$, uniform $h=\pi/4$ $\Rightarrow$ $n=(3\pi/2)/(\pi/4)=6$ subintervals, 7 nodes $x_i=i\pi/4$.
Step 1 — evaluate $f(x)=\sin x$: $\sin0=0$; $\sin\frac\pi4=\tfrac{\sqrt2}2\approx0.70711$; $\sin\frac\pi2=1$; $\sin\frac{3\pi}4=\tfrac{\sqrt2}2$; $\sin\pi=0$; $\sin\frac{5\pi}4=-\tfrac{\sqrt2}2$; $\sin\frac{3\pi}2=-1$.
Step 2 — uniform trapezoid (endpoints half-weighted): $\tilde I=\tfrac h2(f_0{+}f_6)+h\sum_{i=1}^{5}f_i$ with $\sum_{1}^{5}f_i=\tfrac{\sqrt2}2+1+\tfrac{\sqrt2}2+0-\tfrac{\sqrt2}2=1+\tfrac{\sqrt2}2\approx1.70711$.
Step 3 — multiply out: $\tilde I=\tfrac h2(f_0{+}f_6)+h\sum_{1}^{5}f_i$\\
$=\tfrac\pi4\big(\tfrac12(-1)+1.70711\big)=\tfrac\pi4(1.20711)\approx0.78540\cdot1.20711\approx0.94806$.
Step 4 — exact $I=\int_0^{3\pi/2}\sin x\,dx=[-\cos x]_0^{3\pi/2}=-\cos\tfrac{3\pi}2+\cos0=0+1=1$; err $=|1-0.94806|\approx0.05194$.
\wk{How to draw trapezoids (Pract.6):} (1) mark all $n{+}1$ nodes on the axis; (2) plot $(x_i,f(x_i))$; (3) join consecutive points with straight segments; (4) each strip is a trapezoid (drop verticals to the axis). Parabolas (Simpson): same nodes, one arc per pair through its 3 points.

\subsection{Simpson $\int_2^4\tfrac{dx}{\ln x}$ (Quiz 8)}

\subsection{Pseudoinverse $\to$ 1D formulas (HW09)}
$X^TX\bar a=X^T\bar y$ with $X=[1\ x_i]$ gives $\big(\begin{smallmatrix}n{+}1&\sum x_i\\\sum x_i&\sum x_i^2\end{smallmatrix}\big)\big(\begin{smallmatrix}a_0\\a_1\end{smallmatrix}\big)=\big(\begin{smallmatrix}\sum y_i\\\sum x_iy_i\end{smallmatrix}\big)$; dividing by $n{+}1$ and solving (Cramer) returns\\
$a_1=\tfrac{\overline{xy}-\bar x\bar y}{\overline{x^2}-\bar x^2}$, $a_0=\bar y-a_1\bar x$.
Cramer in full: $\Delta=(n{+}1)\sum x_i^2-(\sum x_i)^2$; $a_0=\tfrac{(\sum y_i)(\sum x_i^2)-(\sum x_i)(\sum x_iy_i)}{\Delta}$; $a_1=\tfrac{(n{+}1)(\sum x_iy_i)-(\sum x_i)(\sum y_i)}{\Delta}$; dividing top and bottom by $(n{+}1)^2$ gives the averaged formulas.
\ex{HW09 Q2 setup ($m{=}2,n{=}2$, 3 pts):} $X$ is $3\times3$ (first col of 1's, then $x_1^{(i)},x_2^{(i)}$ rows); $X^+=(X^TX)^{-1}X^T$ is $3\times3$. \ex{Q3:} $X^+X=I$ always; $XX^+=I$ only if $X$ square nonsingular.
